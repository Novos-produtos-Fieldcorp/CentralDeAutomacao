create table public.end_vaga (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  numero numeric null,
  ds_complemento text null,
  st_end boolean null,
  logradouro_id bigint null,
  vaga_id bigint null,
  constraint end_vaga_pkey primary key (id),
  constraint end_vaga_logradouro_id_fkey foreign KEY (logradouro_id) references logradouro (id_logradouro) on update CASCADE on delete set null,
  constraint end_vaga_vaga_id_fkey foreign KEY (vaga_id) references vaga (id) on update RESTRICT on delete set null
) TABLESPACE pg_default;


create table public.st_vaga (
  id bigint generated by default as identity not null,
  status_vaga text null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone null,
  company_id bigint null,
  constraint st_vaga_pkey primary key (id),
  constraint st_vaga_company_id_fkey foreign KEY (company_id) references company (company_id) on update CASCADE on delete set null
) TABLESPACE pg_default;

create table public.operacao (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone null,
  operacao text null,
  company_id bigint null,
  constraint operascao_pkey primary key (id),
  constraint operascao_company_id_fkey foreign KEY (company_id) references company (company_id) on update CASCADE on delete set null
) TABLESPACE pg_default;

create table public.unidade (
  id bigint generated by default as identity not null,
  unidade text null,
  company_id bigint null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone null,
  constraint unidade_pkey primary key (id),
  constraint unidade_company_id_fkey foreign KEY (company_id) references company (company_id) on update CASCADE on delete set null
) TABLESPACE pg_default;

create table public.vaga (
  id bigint generated by default as identity not null,
  nome text null,
  descricao text null,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone null,
  quantidade numeric null,
  dias_trabalho text null,
  horario text null,
  dt_limite timestamp with time zone null,
  company_id bigint null,
  unidade_id bigint null,
  operacao_id bigint null,
  st_vaga_id bigint null,
  cliente_id bigint null,
  gr_id bigint null,
  constraint vaga_pkey primary key (id),
  constraint vaga_company_id_fkey foreign KEY (company_id) references company (company_id) on update CASCADE on delete set null,
  constraint vaga_gr_id_fkey foreign KEY (gr_id) references gr_empresa (id) on update CASCADE on delete set null,
  constraint vaga_cliente_id_fkey foreign KEY (cliente_id) references cliente (cliente_id) on update CASCADE on delete set null,
  constraint vaga_st_vaga_id_fkey foreign KEY (st_vaga_id) references st_vaga (id),
  constraint vaga_unidade_id_fkey foreign KEY (unidade_id) references unidade (id) on update CASCADE on delete set null,
  constraint vaga_operacao_id_fkey foreign KEY (operacao_id) references operacao (id) on update CASCADE on delete set null
) TABLESPACE pg_default;
